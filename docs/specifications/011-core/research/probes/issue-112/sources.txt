751: /**
752:  * The #387 fallback locations, in every shared OS temp root. Kept narrow on
753:  * purpose: exact prefix, 12 lowercase-hex name, directories only, and
754:  * ownership — a directory is removed only when its `.cez-owner` marker
755:  * names THIS dataDir and none of this dataDir's kept runs claims it, or
756:  * when its marker names a dataDir that no longer exists
757:  * (hearsay-tools/cezarion#1911): nothing else will ever remove that one. A
758:  * marker that is missing, unreadable or names a live dataDir is another
759:  * project's, or unknown, and is never ours to remove.
760:  */
761: function sweepFallbackRoots(dataDir: string, keep: ReadonlySet<string>): string[] {
762:   const owner = dataDirOwner(dataDir);
763:   const claimed = new Set(
764:     [...keep].filter(safeRunId).map((id) => basename(fallbackTmpDir(osTempRoot(), dataDir, id))),
765:   );
766:   const reaped: string[] = [];
767:   for (const osRoot of osTempRoots()) {
768:     let fallbackEntries: Dirent[];
769:     try {
770:       fallbackEntries = readdirSync(osRoot, { withFileTypes: true });
771:     } catch {
772:       continue; // an unreadable root is skipped, not fatal: others remain.
773:     }
774:     for (const entry of fallbackEntries) {
775:       if (!fallbackSweepable(osRoot, entry, owner, claimed)) continue;
776:       try {
777:         rmSync(join(osRoot, entry.name), { recursive: true, force: true });
778:         reaped.push(entry.name);
779:       } catch {
780:         // best-effort: a locked directory is retried on the next boot.
781:       }
782:     }
783:   }
784:   return reaped;
785: }

787: function fallbackSweepable(osRoot: string, entry: Dirent, owner: string, claimed: ReadonlySet<string>): boolean {
788:   if (!entry.isDirectory() || !entry.name.startsWith(FALLBACK_PREFIX)) return false;
789:   if (!/^[0-9a-f]{12}$/.test(entry.name.slice(FALLBACK_PREFIX.length))) return false;
790:   let entryOwner: string;
791:   try {
792:     entryOwner = readFileSync(join(osRoot, entry.name, OWNER_FILE), 'utf8').trim();
793:   } catch {
794:     return false; // no readable marker: unknown ownership is never ours to remove.
795:   }
796:   return entryOwner === owner ? !claimed.has(entry.name) : ownerDataDirGone(join(osRoot, entry.name), entryOwner);
797: }
