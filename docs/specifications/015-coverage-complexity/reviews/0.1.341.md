**Trigger and result.** Begin with no exclude. Both `src/a.ts` and the existing `src/gen/b.ts` have judged coverage states. In one pass, set `coverage.exclude` to `src/gen/**` and move a function from `a.ts` into `b.ts`. The repair correctly removes coverage states from `b.ts`, including those of its function that was already there. The coverage step then finds none to count. `readCoverageChange` returns `null`. With no prior change record, the required summary is absent, rather than merely one file short.

**Independent clean-tree proof.** This external probe uses the existing rig and supplies the same new matcher as the daemon. The two parameter values select the ordinary and staged edit paths. Before the move it proves that the existing destination has all three states; after it, complexity alone. The last assertion checks the missing summary record.

import { expect, it } from "vitest";
import { coverageChecks, readCoverageChange } from "../../../src/core/code/crap/index.js";
import { DEFAULT_POLICY } from "../../../src/core/types/index.js";
import { WT } from "../../state/helpers.js";
import { coverageRig, sourceOf } from "./rig.js";

it.each([{}, { partFiles: 1 }])("exclude change plus move: %j", async (part) => {
  const rig = coverageRig();
  await rig.edit({ "src/a.ts": "made 9 m1\n", "src/gen/b.ts": "stay 2 s1\n" }, part);
  const source = (path: string) =>
    sourceOf(path, rig.store.files.list(WT, [path])[0]?.blob ?? "", 1, []);
  rig.loader("test/a.test.ts", {
    closure: ["src/a.ts", "src/gen/b.ts"],
    record: [source("src/a.ts"), source("src/gen/b.ts")],
  });
  rig.step();
  const measures = () =>
    rig.store.states.list(WT, { paths: ["src/gen/b.ts"] }).map((s) => s.measure);
  expect(measures()).toEqual(["complexity", "crap", "uncovered"]);
  const checks = coverageChecks({
    ...DEFAULT_POLICY,
    coverage: { ...DEFAULT_POLICY.coverage, exclude: ["src/gen/**"] },
  });
  await rig.edit(
    { "src/a.ts": "", "src/gen/b.ts": "stay 2 s1\nmade 9 m1\n" },
    { ...part, coverageExcluded: checks.excluded },
  );
  rig.step({ checks });
  expect(measures()).toEqual(["complexity", "complexity"]);
  expect(readCoverageChange(rig.store.meta, WT)).toMatchObject({ excluded: 1 });
});
