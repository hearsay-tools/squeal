This probe uses the existing public test rig. To repeat it, save it as a test beside `test/code/crap/rig.ts` and run that test alone:

```ts
import { expect, it } from "vitest";
import { coverageChecks } from "../../../src/core/code/crap/index.js";
import { DEFAULT_POLICY } from "../../../src/core/types/index.js";
import { WT } from "../../state/helpers.js";
import { coverageRig, sourceOf } from "./rig.js";

it("a move into coverage.exclude keeps only complexity", async () => {
  const rig = coverageRig();
  const checks = coverageChecks({
    ...DEFAULT_POLICY,
    coverage: { ...DEFAULT_POLICY.coverage, exclude: ["src/gen/**"] },
  });
  await rig.edit({ "src/a.ts": "made 9 m1\n" });
  const blob = rig.store.files.list(WT, ["src/a.ts"])[0]?.blob ?? "";
  rig.loader("test/a.test.ts", {
    closure: ["src/a.ts"],
    record: [sourceOf("src/a.ts", blob, 1, [])],
  });
  rig.step({ checks });
  expect(rig.states("crap")).toEqual(["made fail 90"]);
  await rig.edit({ "src/a.ts": "", "src/gen/b.ts": "made 9 m1\n" });
  rig.step({ checks });
  expect(
    rig.store.states.list(WT, { paths: ["src/gen/b.ts"] }).map((s) => s.measure),
  ).toEqual(["complexity"]);
});
```
