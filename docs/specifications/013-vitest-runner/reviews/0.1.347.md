## Bounded actual-Vitest-4 proof

Run at the clean candidate with `node --import tsx --input-type=module` and the following stdin program. It creates two temporary client projects, uses Cezar's already installed Vitest 4.1.10 / Vite 8.1.4, and removes all temporary files. It does not execute Cezar's tests or edit its checkout.

```js
import assert from "node:assert/strict";
import { mkdtempSync, mkdirSync, writeFileSync, symlinkSync,
  copyFileSync, readFileSync, rmSync } from "node:fs";
import { dirname, join } from "node:path";
import { tmpdir } from "node:os";
import { createVitestAdapter } from "./src/runners/vitest/index.ts";
import { buildRecord, coverageProvider } from "./src/core/code/coverage/index.ts";
const dir = mkdtempSync(join(tmpdir(), "sq-review-v4-"));
const root = join(dir, "workspace");
const write = (file, text) => {
  const path = join(root, file);
  mkdirSync(dirname(path), { recursive: true });
  writeFileSync(path, text);
};
let adapter;
try {
  write("package.json", '{"type":"module"}');
  write("vitest.config.ts",
    'export default {test:{projects:["./a/vitest.config.ts","./b/vitest.config.ts"]}}');
  symlinkSync("/home/agent/projects/squeal/.ai/cezar/t/6b240e54/repro/cezar/node_modules",
    join(root, "node_modules"), "dir");
  for (const name of ["a", "b"]) {
    write(`${name}/vitest.config.ts`,
      `export default {test:{name:"${name}",include:["test/*.test.ts"],environment:"./client-env.ts",maxWorkers:1}}`);
    write(`${name}/client-env.ts`,
      'export default {name:"client-env",viteEnvironment:"client",setup(){return {teardown(){}}}}');
    write(`${name}/src/value.ts`, `export function value(){ return "${name}"; }`);
    write(`${name}/test/value.test.ts`,
      `import {it,expect} from "vitest"; import {value} from "../src/value.ts"; it("own value",()=>expect(value()).toBe("${name}"));`);
  }
  const provider = join(dir, "provider.mjs");
  copyFileSync(coverageProvider(), provider);
  const runs = [];
  adapter = await createVitestAdapter({ root, note: () => {}, coverage: {
    enabled: () => true, provider, tempDir: join(dir, "payloads"),
    collected: run => runs.push(run),
  }});
  await adapter.open?.();
  const requested = ["a", "b"].map(project => ({
    project, path: `${project}/test/value.test.ts`,
  }));
  const report = await adapter.run(requested, {
    runId: "review-v4", logDir: join(dir, "logs"), timeoutMs: 30000,
  });
  assert.equal(report.failure, null);
  assert.equal(report.end, "completed");
  assert.deepEqual(report.results.map(x => x.outcome), ["pass", "pass"]);
  const kept = runs.flatMap(run => run.files)
    .sort((a, b) => a.testFile.project.localeCompare(b.testFile.project));
  assert.deepEqual(kept.map(f => f.testFile), requested);
  for (const file of kept) {
    const payload = JSON.parse(readFileSync(file.payload, "utf8"));
    const record = buildRecord(file, payload.result, "sha1", new Map());
    assert.deepEqual(record.sources.map(s => s.path), [
      `${file.testFile.project}/src/value.ts`,
      `${file.testFile.project}/test/value.test.ts`,
    ]);
    assert.ok(record.sources[0].functions.some(f => f[4] === 1));
  }
  console.log("PASS: actual Vitest 4.1.10 + Vite 8.1.4; two separate client project servers; equal relative test names; both tests pass and both records map to their own executed sources with function hits");
} finally {
  await adapter?.close();
  rmSync(dir, { recursive: true, force: true });
}
```
